import type { Metadata } from 'next';
import { JOBS_SOURCE } from '@/lib/jobs/config';
import { fetchJobs } from '@/lib/jobs/feed';
import JobCard from '@/components/jobs/JobCard';
import JobFilters from '@/components/jobs/JobFilters';
import PlacementPartnerEmbed from '@/components/jobs/PlacementPartnerEmbed';
import Container from "@/components/site/Container";
import PageHeader from "@/components/site/PageHeader";
import ClosingCta from "@/components/site/ClosingCta";
import Link from "next/link";

export const revalidate = 3600;

export const metadata: Metadata = {
  title: 'Job Vacancies — H&S Labour Brokers South Africa',
  description: 'Find permanent, contract and temporary employment services (TES) jobs across Johannesburg, Cape Town, Durban, Pretoria and Gqeberha. H&S Labour Brokers connects skilled candidates with top employers.',
};

export default async function JobsPage({
  searchParams,
}: {
  searchParams: Promise<{ city?: string; category?: string; sort?: string }>;
}) {
  const params = await searchParams;

  // ---- IFRAME BRANCH ----
  if (JOBS_SOURCE === 'iframe') {
    return (
      <>
        <PageHeader
          eyebrow="Careers"
          title="Current vacancies"
          intro="H&S Labour Brokers supplies permanent, contract and TES candidates across South Africa — construction, engineering, manufacturing, logistics, facilities, cleaning, security and healthcare, with deep coverage in Johannesburg, Cape Town, Durban, Pretoria and Gqeberha."
          imageSrc="/images/careers-people.jpg"
          imageAlt="Candidates placed by H&S Labour Brokers"
        />
        <section className="bg-white py-16 sm:py-20">
          <Container>
            <PlacementPartnerEmbed />
            <noscript>
              <div className="mt-8 text-center text-sm text-slate-600">
                JavaScript is required to view the full job board.{' '}
                <Link
                  href="https://hs-labour-brokers.placementpartner.co.za/"
                  target="_blank"
                  rel="noopener noreferrer"
                  className="text-green-dark underline"
                >
                  Open the job board in a new tab
                </Link>
              </div>
            </noscript>
          </Container>
        </section>
        <ClosingCta />
      </>
    );
  }

  // ---- FEED BRANCH ----
  const jobs = await fetchJobs();

  // Extract unique cities and categories
  const cities = [...new Set(jobs.map((job) => job.city).filter(Boolean))].sort();
  const categories = [
    ...new Set(jobs.flatMap((job) => job.categories ?? [])),
  ].sort();

  // Active filter values from URL
  const activeCity = params.city ?? '';
  const activeCategory = params.category ?? '';
  const activeSort = params.sort ?? 'date';

  // Apply filters & sorting
  const filteredJobs = jobs
    .filter((job) => !activeCity || job.city === activeCity)
    .filter((job) => !activeCategory || (job.categories ?? []).includes(activeCategory))
    .sort((a, b) => {
      if (activeSort === 'title') {
        return a.title.localeCompare(b.title);
      }
      // default: date descending
      return new Date(b.date).getTime() - new Date(a.date).getTime();
    });

  return (
    <>
      <PageHeader
        eyebrow="Careers"
        title="Job vacancies — South Africa"
        intro="Browse permanent, contract and temporary roles placed by H&S Labour Brokers across Johannesburg, Cape Town, Durban, Pretoria, Gqeberha and surrounding areas."
        imageSrc="/images/careers-people.jpg"
        imageAlt="Candidates placed by H&S Labour Brokers"
      />

      <section className="bg-white py-16 sm:py-20">
        <Container>
          {jobs.length > 0 && (
            <JobFilters cities={cities} categories={categories} />
          )}

          {filteredJobs.length > 0 ? (
            <div className="mt-8 grid grid-cols-1 gap-6 md:grid-cols-2 lg:grid-cols-3">
              {filteredJobs.map((job) => (
                <JobCard key={job.id ?? job.slug} job={job} />
              ))}
            </div>
          ) : (
            <div className="mt-8 flex flex-col items-center justify-center rounded-2xl bg-slate-50 px-6 py-16 text-center">
              <h2 className="text-xl font-semibold text-navy">
                No jobs match your filters
              </h2>
              <p className="mt-2 text-slate-600">
                Try adjusting your search or clearing all filters.
              </p>
              <Link
                href="/jobs"
                className="mt-4 text-green-dark underline hover:text-green"
              >
                Clear all filters
              </Link>
            </div>
          )}
        </Container>
      </section>

      <ClosingCta />
    </>
  );
}
